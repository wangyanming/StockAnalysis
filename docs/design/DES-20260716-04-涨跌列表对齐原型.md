### 1.4 工具函数

```javascript
// 金额格式化：≥1亿显示亿，否则万
function fmtMoney(v) {
  v = parseFloat(v) || 0;
  if (v >= 100000000) return (v / 100000000).toFixed(2) + '亿';
  return (v / 10000).toLocaleString(undefined, {minimumFractionDigits:0, maximumFractionDigits:0}) + '万';
}

// 时间格式化：093039 → 09:30
function fmtTime(t) {
  if (!t) return '-';
  const s = String(t);
  if (s.length >= 4) return s.slice(0,2) + ':' + s.slice(2,4);
  return s;
}
```
